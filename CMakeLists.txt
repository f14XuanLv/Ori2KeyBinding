cmake_minimum_required(VERSION 3.24)
project(OriPadMod VERSION 0.1.2 LANGUAGES C CXX ASM_MASM)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
add_compile_options($<$<COMPILE_LANGUAGE:CXX>:/utf-8>)
add_subdirectory(vendor/minhook EXCLUDE_FROM_ALL)
add_library(imgui STATIC vendor/imgui/imgui.cpp vendor/imgui/imgui_draw.cpp
    vendor/imgui/imgui_tables.cpp vendor/imgui/imgui_widgets.cpp
    vendor/imgui/backends/imgui_impl_dx11.cpp)
target_include_directories(imgui PUBLIC vendor/imgui)
add_library(OriPadMod SHARED src/runtime.cpp src/overlay.cpp src/version_proxy.cpp src/version_trampolines.asm src/version.def)
target_include_directories(OriPadMod PRIVATE src)
target_compile_options(OriPadMod PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4>)
target_link_libraries(OriPadMod PRIVATE minhook imgui d3d11 dxgi user32)
set_target_properties(OriPadMod PROPERTIES OUTPUT_NAME version)
enable_testing()
add_executable(OriPadTests tests/mapping_tests.cpp)
target_include_directories(OriPadTests PRIVATE src)
add_test(NAME mapping COMMAND OriPadTests)
add_executable(OriPadHarness WIN32 tests/harness.cpp)
target_link_libraries(OriPadHarness PRIVATE d3d11 dxgi user32)
